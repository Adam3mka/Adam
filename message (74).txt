local function sh(ch,on)

he.Transparency=t

if d:IsA("Decal")then d.Transparency=t end
end

if o:IsA("Accessory")then
local n=o.Name:lower()
local hd=o:FindFirstChild("Handle")
if hd and(n:find("face")or n:find("mask"))then
hd.Transparency=t
end
end
end
end

local function wa(ch,ac)
local hd=ac:FindFirstChild("Handle")
if not hd or not hd:IsA("BasePart")then return end
if hd:FindFirstChild("AccessoryWeld")then return end
for _,x in ipairs(ac:GetDescendants())do
if x:IsA("BasePart")then
x.Anchored=false
x.CanCollide=false
x.Massless=true
end
end
local aa,tp,tc
for _,a in ipairs(hd:GetChildren())do
if a:IsA("Attachment")then
for _,p in ipairs(ch:GetChildren())do
if p:IsA("BasePart")then
local ca=p:FindFirstChild(a.Name)
if ca and ca:IsA("Attachment")then
aa,tp,tc=a,p,ca.CFrame
break
end
end
end
if not aa then
local fb=AF[a.Name]
local p=fb and ch:FindFirstChild(fb[1])
if p then
aa,tp,tc=a,p,fb[2]
end
end
if aa then break end
end
end
local w=Instance.new("Weld")
w.Name="AccessoryWeld"
if aa then
w.Part0=hd
w.Part1=tp
w.C0=aa.CFrame
w.C1=tc
else
local he=ch:FindFirstChild("Head")
if not he then return end
w.Part0=he
w.Part1=hd
w.C0=CFrame.new(0,0.5,0)
w.C1=ac.AttachmentPoint
end
w.Parent=hd
end
local function ab(ch,ap)
local bc=ap:FindFirstChildOfClass("BodyColors")
if bc then
local cs={
["Head"]=bc.HeadColor3,
["Torso"]=bc.TorsoColor3,
["Left Arm"]=bc.LeftArmColor3,
["Right Arm"]=bc.RightArmColor3,
["Left Leg"]=bc.LeftLegColor3,
["Right Leg"]=bc.RightLegColor3,
}
for n,c in pairs(cs)do
local pt2=ch:FindFirstChild(n)
if pt2 and pt2:IsA("BasePart")then
pt2.Color=c
end
end
end
local th=ap:FindFirstChild("Head")
local mh=ch:FindFirstChild("Head")
if th and mh then
local nm=th:FindFirstChildOfClass("SpecialMesh")
if nm then
local om=mh:FindFirstChildOfClass("SpecialMesh")
if om then om:Destroy()end
nm:Clone().Parent=mh
end
local nf=th:FindFirstChildOfClass("Decal")
if nf then
for _,d in ipairs(mh:GetChildren())do
if d:IsA("Decal")then d:Destroy()end
end
nf:Clone().Parent=mh
end
end
end
local function ao(uid,ch)
local ok,ap=pcall(function()
return P:GetCharacterAppearanceAsync(uid)
end)
if not ok or not ap then return false end
co(ch)
for _,o in ipairs(ap:GetChildren())do
if o:IsA("Clothing")or o:IsA("ShirtGraphic")or o:IsA("BodyColors")or o:IsA("CharacterMesh")then
o:Clone().Parent=ch
elseif o:IsA("Accessory")then
local cl=o:Clone()
cl.Parent=ch
local okw,er=pcall(wa,ch,cl)
if not okw then warn("Weld failed for "..cl.Name..": "..tostring(er))end
end
end
pcall(ab,ch,ap)
ap:Destroy()
if fk then ak(ch)end
if fh then sh(ch,true)end
return true
end
